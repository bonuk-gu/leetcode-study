# 18. 4Sum

https://leetcode.com/problems/4sum/description/

## 문제

Given an array nums of n integers, return an array of all the unique quadruplets [nums[a], nums[b], nums[c], nums[d]] such that:

- 0 <= a, b, c, d < n
- a, b, c, and d are distinct.
- nums[a] + nums[b] + nums[c] + nums[d] == target

You may return the answer in any order.

- Example 1:
  Input: nums = [1,0,-1,0,-2,2], target = 0
  Output: [[-2,-1,1,2],[-2,0,0,2],[-1,0,0,1]]

- Example 2:
  Input: nums = [2,2,2,2,2], target = 8
  Output: [[2,2,2,2]]

## 내 풀이(TS)

```ts
function fourSum(nums: number[], target: number): number[][] {
  const output: number[][] = [];
  if (nums.length < 4) {
    return output;
  }

  nums.sort((a, b) => a - b);

  if (
    target < nums[0] + nums[1] + nums[2] + nums[3] ||
    target >
      nums[nums.length - 1] +
        nums[nums.length - 2] +
        nums[nums.length - 3] +
        nums[nums.length - 4]
  ) {
    return output;
  }

  for (let i: number = 0; i < nums.length - 3; i++) {
    if (i >= 1 && nums[i] === nums[i - 1]) {
      continue;
    }

    const target2: number = target - nums[i];

    for (let j: number = i + 1; j < nums.length - 2; j++) {
      if (j > i + 1 && nums[j] === nums[j - 1]) {
        continue;
      }

      const target3: number = target2 - nums[j];
      let left: number = j + 1;
      let right: number = nums.length - 1;

      while (left < right) {
        if (nums[left] + nums[right] === target3) {
          output.push([nums[i], nums[j], nums[left], nums[right]]);
          left++;
          right--;
          while (left < right && nums[left] === nums[left - 1]) {
            left++;
          }
          while (left < right && nums[right] === nums[right + 1]) {
            right--;
          }
        } else if (nums[left] + nums[right] < target3) {
          left++;
          while (left < right && nums[left] === nums[left - 1]) {
            left++;
          }
        } else {
          right--;
          while (left < right && nums[right] === nums[right + 1]) {
            right--;
          }
        }
      }
    }
  }

  return output;
}
```

- 시간복잡도: O(n^3)
  정렬: O(n log g)
  i: O(n)
  j: O(n)
  left/right: O(n)
- 공간복잡도: O(1) (배열 output 제외)

## 개선된 코드

```ts
function fourSum(nums: number[], target: number): number[][] {
  const result: number[][] = [];
  const n = nums.length;

  if (n < 4) {
    return result;
  }

  nums.sort((a, b) => a - b);

  for (let i = 0; i < n - 3; i++) {
    if (i > 0 && nums[i] === nums[i - 1]) {
      continue;
    }

    // 현재 i에서 만들 수 있는 최소값
    const minSum = nums[i] + nums[i + 1] + nums[i + 2] + nums[i + 3];

    if (minSum > target) {
      break;
    }

    // 현재 i에서 만들 수 있는 최대값
    const maxSum = nums[i] + nums[n - 1] + nums[n - 2] + nums[n - 3];

    if (maxSum < target) {
      continue;
    }

    for (let j = i + 1; j < n - 2; j++) {
      if (j > i + 1 && nums[j] === nums[j - 1]) {
        continue;
      }

      // 현재 i, j에서 만들 수 있는 최소값
      const minSum = nums[i] + nums[j] + nums[j + 1] + nums[j + 2];

      if (minSum > target) {
        break;
      }

      // 현재 i, j에서 만들 수 있는 최대값
      const maxSum = nums[i] + nums[j] + nums[n - 1] + nums[n - 2];

      if (maxSum < target) {
        continue;
      }

      let left = j + 1;
      let right = n - 1;

      while (left < right) {
        const sum = nums[i] + nums[j] + nums[left] + nums[right];

        if (sum === target) {
          result.push([nums[i], nums[j], nums[left], nums[right]]);

          left++;
          right--;

          while (left < right && nums[left] === nums[left - 1]) {
            left++;
          }

          while (left < right && nums[right] === nums[right + 1]) {
            right--;
          }
        } else if (sum < target) {
          left++;
        } else {
          right--;
        }
      }
    }
  }

  return result;
}
```

## Python 풀이코드

```py
class Solution:
    def fourSum(self, nums: List[int], target: int) -> List[List[int]]:
        output = []

        if len(nums) < 4:
            return output

        nums.sort()

        if target < nums[0] + nums[1] + nums[2] + nums[3] or target > nums[-1] + nums[-2] + nums[-3] + nums[-4]:
            return output

        i = 0
        while i < len(nums) - 3:
            if target < nums[i] + nums[i + 1] + nums[i + 2] + nums[i + 3]:
                break
            if target > nums[i] + nums[-1] + nums[-2] + nums[-3]:
                i += 1
                continue
            if i >= 1 and nums[i] == nums[i-1]:
                i += 1
                continue

            j = i + 1
            while j < len(nums) - 2:
                if target < nums[i] + nums[j] + nums[j + 1] + nums[j + 2]:
                    break
                if target > nums[i] + nums[j] + nums[-1] + nums[-2]:
                    j += 1
                    continue
                if j > i + 1 and nums[j] == nums[j-1]:
                    j += 1
                    continue

                left = j + 1
                right = len(nums) - 1

                while left < right:
                    four_sum = nums[i] + nums[j] + nums[left] + nums[right]
                    if four_sum == target:
                        output.append([nums[i], nums[j], nums[left], nums[right]])

                        left += 1
                        right -= 1

                        while left < right and nums[left] == nums[left - 1]:
                            left += 1
                        while left < right and nums[right] == nums[right + 1]:
                            right -= 1
                    elif four_sum < target:
                        left += 1
                    else:
                        right -= 1
                j += 1

            i += 1

        return output
```
